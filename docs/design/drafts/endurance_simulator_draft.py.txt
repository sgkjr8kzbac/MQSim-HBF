"""Endurance of KV cache stored on HBF (module M4): closed-form steady state and event simulation.

``analytic`` gives instant steady-state curves; ``simulate`` runs a request-level simulation at
segment granularity (a segment is one block index across the banks of a partition: a superblock
of 1,024 banks, a channel block of 64 or a die block of 16). Both take simulated time from a
``Clock``: the phase A per-layer estimate est = layer_ns + round_ns x rounds (fitted to MQSim,
design-B F1) fed into the unchanged ``system.evaluate``.

Spec rules used (OCP HBF v0.7.0):

* Writes are sequential within a NAND block, no skipping (section 5.4.1 items 4 and 9); writing
  page 0 auto-erases the block (item 5). Every page-0 write counts as one P/E cycle, also on a
  never-written block (assumption: spec-literal and conservative). There is no garbage collection
  (section 11.4): space is reclaimed only by the host (evict, compact or rewrite).
* Read disturb: the host refreshes a block after X reads (section 11.5, product specific); the
  base die keeps a per-block page-read counter (section 4.4). The counter is assumed to reset at
  erase.
* Retention: 24 h power-on (section 9 Table 33); refresh "typically every 24-48 hours" (11.5).
* Replay (section 5.7) rewrites a block from page 0 to its last written page in place.

``python -m hbf.endurance --model ... --workload long32k --concurrency 8 --hours 24 --output DIR``
runs the simulation; ``--analytic`` writes the closed-form rows instead. ``build_parser``,
``configure``, ``run``, ``summarize_run`` and ``METRICS`` mirror ``hbf.cli`` for the sweep driver.
"""

# KV geometry

class Geometry:
    """Stored KV of one request: pages per context length, per-layer read demand.

    ``stored_pages`` counts full pages only (a partial last page stays in HBM staging);
    ``read_pages`` counts every page a dense step senses (ceil), as phase A does.
    """

    def __init__(self, layers, heads, tensors, page_bytes=4096):
        self.layers, self.heads, self.page_bytes = layers, heads, page_bytes
        self.tensors = tuple(tensors)
        self.hbf = tuple(spec for spec in self.tensors if spec.tier == "hbf")
        self.hbm = tuple(spec for spec in self.tensors if spec.tier == "hbm")
        if not self.hbf:
            raise ValueError("At least one tensor must be stored on HBF")
        self.tpp = tuple(page_bytes // spec.token_bytes for spec in self.hbf)
        if 0 in self.tpp:
            raise ValueError("A token row must fit in one page")
        self.data = tuple(spec for spec in self.tensors if spec.role == "data")

    @property
    def token_bytes(self):
        """KV bytes per token of all layers and heads (all tiers)."""
        return self.layers * self.heads * sum(spec.token_bytes for spec in self.tensors)

    @property
    def hbf_token_bytes(self):
        return self.layers * self.heads * sum(spec.token_bytes for spec in self.hbf)

    @staticmethod
    def window(spec, ctx):
        ctx = np.asarray(ctx, dtype=np.int64)
        lo, hi = spec.window or (None, None)
        lo = np.zeros_like(ctx) if lo is None else (ctx + lo if lo < 0 else np.full_like(ctx, lo))
        hi = ctx.copy() if hi is None else (ctx + hi if hi < 0 else np.full_like(ctx, hi))
        lo = np.clip(lo, 0, ctx)
        return lo, np.clip(hi, lo, ctx)

    def stored_pages(self, ctx):
        """Full HBF pages of a request with ``ctx`` tokens (all layers and heads)."""
        ctx = np.asarray(ctx, dtype=np.int64)
        total = np.zeros_like(ctx)
        for spec, tpp in zip(self.hbf, self.tpp):
            lo, hi = self.window(spec, ctx)
            total += (hi - lo) // tpp
        return self.layers * self.heads * total

    def row_pages(self, ctx, role_share=None):
        """Pages one layer senses per step (ceil), summed over requests, scaled per role."""
        ctx = np.asarray(ctx, dtype=np.int64)
        total = 0.0
        for spec, tpp in zip(self.hbf, self.tpp):
            lo, hi = self.window(spec, ctx)
            share = 1.0 if role_share is None else role_share.get(spec.role, 1.0)
            total += float(np.sum(-(-(hi - lo) // tpp))) * share
        return self.heads * total

    def partial_pages(self):
        """Pages of HBM staging per running request: one partial page per (layer, head, HBF tensor)."""
        return self.layers * self.heads * len(self.hbf)

    def demand(self, ctx, hbm_tokens=None):
        """Per-layer demand of one dense step over requests with ``ctx`` tokens (phase A columns)."""
        ctx = np.asarray(ctx, dtype=np.int64)
        extra = np.zeros_like(ctx) if hbm_tokens is None else np.asarray(hbm_tokens, dtype=np.int64)
        burst = hbm = elements = dequant = 0.0
        for spec in self.tensors:
            lo, hi = self.window(spec, ctx)
            tokens = hi - lo
            if spec.tier == "hbf":
                full = tokens * spec.token_bytes // self.page_bytes
                rest = tokens * spec.token_bytes - full * self.page_bytes
                burst += float(np.sum(full * self.page_bytes + -(-rest // 64) * 64))
            else:
                hbm += float(np.sum(tokens)) * spec.token_bytes
            if spec.role == "data":
                elements += float(np.sum(tokens)) * spec.dims
            dequant += float(np.sum(tokens)) * spec.dims * spec.dequant_ops
        hbm += float(np.sum(extra)) * sum(spec.token_bytes for spec in self.hbf)
        los, his = zip(*(self.window(spec, ctx) for spec in self.data)) if self.data else ((ctx * 0,), (ctx,))
        attended = float(np.sum(np.max(his, axis=0) - np.min(los, axis=0)))
        h = self.heads
        return {"burst_bytes": h * burst, "hbm_bytes": h * hbm, "attended_tokens": h * attended,
                "attended_elements": h * elements, "dequant_ops": h * dequant}

# Clock

@dataclass(frozen=True)
class StepTime:
    tpot_ns: float
    gpu_ns: float
    hbm_tpot_ns: float
    h_ns: float  # HBF read time of one layer
    gaps_ns: tuple  # HBF-idle time before each layer's read (sums to tpot - layers x h)
    read_pages: float  # pages one layer senses
    rounds: int
    batch: int


class Clock:
    """Simulated decode step time of a running set (per stack), from phase A's fitted estimate.

    est per layer = layer_ns + round_ns x rounds, rounds = ceil(kappa x pages / banks) (or the
    busiest partition), fed as identical layer rows into ``system.evaluate`` with compute batch
    b x stacks and analysis batch b. ``ratios`` scale the dense demand for a sparse mode
    (pages, attended tokens and elements, burst, HBM bytes, dequant), from ``calibrate``.
    ``rounds_fn(b, ctx)``, when given, returns exact rounds for b equal contexts (phase A layout).
    """

    def __init__(self, model, gpu, geometry, options=None, *, read_ns=4096, banks=1024, stacks=1,
                 layer_ns=F1["layer_ns"], round_ns=F1["round_ns"], balance=1.0, ratios=None, selection="static",
                 kv_format="bf16", rounds_fn=None):
        from .system import SystemOptions
        self.model, self.gpu, self.geometry = model, gpu, geometry
        self.options = options or SystemOptions()
        self.read_ns, self.banks, self.stacks = read_ns, banks, stacks
        self.layer_ns, self.round_ns, self.balance = layer_ns, round_ns, balance
        self.ratios = dict(ratios or {})
        self.selection, self.kv_format, self.rounds_fn = selection, kv_format, rounds_fn
        self.cache = {}

    def describe(self):
        return {"layer_ns": self.layer_ns, "round_ns": self.round_ns, "balance": self.balance,
                "ratios": self.ratios, "read_ns": self.read_ns, "banks": self.banks, "stacks": self.stacks,
                "selection": self.selection, "kv_format": self.kv_format, "overlap": self.options.overlap}

    def step(self, contexts, partitions=None, partition_banks=None, hbm_tokens=None):
        """StepTime of one decode step, or None when nothing runs."""
        ctx = np.asarray(contexts, dtype=np.int64)
        b = len(ctx)
        if not b:
            return None
        geo, r = self.geometry, self.ratios
        hbf_ctx = ctx if hbm_tokens is None else ctx - np.asarray(hbm_tokens, dtype=np.int64)
        share = r.get("pages", 1.0)
        pages = geo.row_pages(hbf_ctx) * share
        if partitions is not None and partition_banks:
            parts = np.asarray(partitions)
            loads = [geo.row_pages(hbf_ctx[parts == p]) * share for p in np.unique(parts)]
            rounds = max(math.ceil(self.balance * load / partition_banks - 1e-9) for load in loads)
        elif self.rounds_fn is not None and len(set(ctx.tolist())) == 1 and hbm_tokens is None:
            rounds = self.rounds_fn(b, int(ctx[0]))
        else:
            rounds = math.ceil(self.balance * pages / self.banks - 1e-9)
        d = geo.demand(hbf_ctx, None if hbm_tokens is None else hbm_tokens)
        scale = {"burst_bytes": "burst", "hbm_bytes": "hbm", "attended_tokens": "attended",
                 "attended_elements": "elements", "dequant_ops": "dequant"}
        d = {key: value * r.get(scale[key], 1.0) for key, value in d.items()}
        key = (b, rounds) + tuple(int(round(math.log1p(d[k]) * 200)) for k in sorted(d))
        hit = self.cache.get(key)
        if hit is not None:
            return hit
        from .system import evaluate
        est = self.layer_ns + self.round_ns * rounds if rounds else 0.0
        rows = [{"step": 0, "layer": layer, "est_ns": est, "bound_ns": rounds * self.read_ns} | d
                for layer in range(self.model.layers)]
        meta = {"batch": b * self.stacks, "stacks": self.stacks, "analysis_batch": b, "selection": self.selection,
                "kv_format": self.kv_format}
        result = evaluate(rows, self.model, self.gpu, self.options, meta)
        row, layers = result.steps[0], result.layers
        attn = [x["attn_staged_ns" if self.options.staged else "attn_ns"] for x in layers]
        h = [est] * len(layers)
        gaps = hbf_gaps(h, [x["pre_ns"] for x in layers], attn, [x["post_ns"] for x in layers],
                        self.options.overlap, self.options.staged)
        gaps[0] += max(0.0, row["tpot_hbf_est_ns"] - sum(h) - sum(gaps))
        out = StepTime(row["tpot_hbf_est_ns"], row["gpu_ns"], row["tpot_hbm_ns"], est, tuple(gaps), pages, rounds, b)
        self.cache[key] = out
        return out


class ConstantClock:
    """Fixed step time (tests and ablations): every step takes ``tpot_ns``, HBF idle half of it."""

    def __init__(self, tpot_ns, layers=1, idle_share=.5):
        self.tpot_ns, self.layers, self.idle_share = tpot_ns, layers, idle_share
        self.stacks = 1

    def describe(self):
        return {"constant_tpot_ns": self.tpot_ns}

    def step(self, contexts, partitions=None, partition_banks=None, hbm_tokens=None):
        if not len(contexts):
            return None
        gap = self.tpot_ns * self.idle_share / self.layers
        h = self.tpot_ns * (1 - self.idle_share) / self.layers
        return StepTime(self.tpot_ns, self.tpot_ns, self.tpot_ns, h, (gap,) * self.layers, 0.0, 0, len(contexts))


def layout_rounds(model, kv_format=None, layout="striped_contiguous", pages_per_block=256):
    """rounds_fn for Clock: exact phase A rounds of b equal requests (layer-major, dense) from the layout."""
    from .access import LAYOUTS, Shape
    from .config import load
    from .formats import parse_format
    cfg = load("ocp", pages_per_block=pages_per_block, blocks_per_plane=max(1, 131072 // pages_per_block))
    tensors = parse_format(kv_format or "bf16", model.head_dim).tensors
    cache = {}

    def rounds(b, ctx):
        if (b, ctx) not in cache:
            lay = LAYOUTS[layout](cfg, Shape(model.layers, model.kv_heads * b, model.head_dim, ctx), tensors=tensors)
            layers = [0] if layout.startswith("striped") else range(model.layers)
            worst = 0
            for layer in layers:
                lpas = [lay.lpa_array(layer, np.repeat(np.arange(lay.shape.kv_heads), lay.block_counts[t]),
                                      np.full(lay.shape.kv_heads * lay.block_counts[t], t),
                                      np.tile(np.arange(lay.block_counts[t]), lay.shape.kv_heads))
                        for t in range(len(lay.tensors))]
                worst = max(worst, int(np.bincount(np.concatenate(lpas) % cfg.units).max()))
            cache[b, ctx] = worst
        return cache[b, ctx]
    return rounds

def _csv(path):
    if not path.is_file():
        return []
    with path.open(newline="") as stream:
        rows = [row for row in csv.DictReader(stream) if row.get("group", "1") == "1"]
    out = []
    for row in rows:
        parsed = {}
        for key, value in row.items():
            try:
                parsed[key] = float(value) if value not in ("", None) else None
            except ValueError:
                parsed[key] = value
        out.append(parsed)
    return out


def phase_a(directory, model, gpu, kv_format, batch, context, *, mode="dense", generation=None, device=(),
            alloc="layer_major", partition="none"):
    """Run one analysis-only phase A pipeline (``hbf.cli``) into ``directory``; returns its CSV rows."""
    from .cli import build_parser, configure, pipeline
    argv = HBF_FLAGS + ["--layout", "striped_contiguous", "--model", model, "--gpu", gpu,
                        "--format", kv_format or "bf16", "--mode", mode, "--context", str(context),
                        "--batch", str(batch), "--analysis-only", "--timing", "--output", str(directory)]
    for key, value in (generation or {}).items():
        argv += [f"--{key.replace('_', '-')}", str(value)]
    argv += list(device)
    if alloc != "layer_major" or partition != "none":
        argv += ["--alloc", alloc, "--partition", partition]
    parser = build_parser()
    args = parser.parse_args(argv)
    run = configure(args)
    saved, sys.argv = sys.argv, ["hbf.cli", *argv]
    try:
        pipeline(directory=directory, binary=None, **run)
    finally:
        sys.argv = saved
    return {"layers": _csv(directory / "analysis_layers.csv"), "steps": _csv(directory / "analysis_steps.csv"),
            "system": _csv(directory / "system_steps.csv")}


def calibrate(directory, model, gpu="b200", kv_format="bf16", *, mode="dense", generation=None, context=32768,
              batches=(1, 8), device=(), policy="packed", partition="none", pages_per_block=256):
    """Fit the Clock to at most three phase A analysis-only runs, cached by a hash of the inputs.

    Two dense layer-major points (``batches``) refit layer_ns and round_ns for the actual device
    flags; one snapshot of the policy at the larger batch in the run's mode gives kappa (rounds
    over ceil(pages / banks)), the demand ratios of the mode against dense, and the per-step hot
    factor (max block reads over mean block reads). Returns a dict for ``Clock(**...)`` fields.
    """
    inputs = {"model": model, "gpu": gpu, "format": kv_format, "mode": mode, "generation": generation or {},
              "context": context, "batches": list(batches), "device": list(device), "policy": policy,
              "partition": partition, "pages_per_block": pages_per_block, "version": 1}
    digest = hashlib.sha256(json.dumps(inputs, sort_keys=True).encode()).hexdigest()[:16]
    root = Path(directory) / f"calib-{digest}"
    cached = root / "calibration.json"
    if cached.is_file():
        return json.loads(cached.read_text())
    root.mkdir(parents=True, exist_ok=True)
    device = list(device) + ["--pages-per-block", str(pages_per_block),
                             "--blocks-per-plane", str(max(1, 131072 // pages_per_block))]
    points, dense = [], {}
    for b in batches:
        target = root / f"dense-b{b}"
        if target.exists():
            import shutil
            shutil.rmtree(target)
        dense[b] = phase_a(target, model, gpu, kv_format, b, context, device=device)
        for row in dense[b]["layers"]:
            if row["pages"]:
                points.append((row["rounds"], row["est_ns"]))
    rounds = np.array([p[0] for p in points], dtype=float)
    est = np.array([p[1] for p in points], dtype=float)
    if len(set(rounds.tolist())) >= 2:
        round_ns, layer_ns = np.polyfit(rounds, est, 1)
    else:
        round_ns = F1["round_ns"]
        layer_ns = float(np.mean(est - round_ns * rounds))
    b = max(batches)
    snap_policy = {"hbm_decode": "packed"}.get(policy, policy)
    snap_policy = snap_policy if snap_policy in ("packed", "dedicated", "vllm_pool") else "layer_major"
    snap_part = partition if snap_policy in ("packed", "dedicated") else "none"
    if mode == "dense" and snap_policy == "layer_major":
        snapshot = dense[b]
    else:
        target = root / f"snapshot-{mode}-{snap_policy}-{snap_part}-b{b}"
        if target.exists():
            import shutil
            shutil.rmtree(target)
        snapshot = phase_a(target, model, gpu, kv_format, b, context, mode=mode, generation=generation,
                           device=device, alloc=snap_policy, partition=snap_part)
    banks = PARTITIONS[snap_part] if snap_part != "none" else 1024
    kappas = [row["rounds"] / math.ceil(row["pages"] / 1024) for row in snapshot["layers"] if row["pages"]]
    kappa = float(np.mean(kappas)) if snap_part == "none" else float(np.mean(kappas))
    sums = lambda rows, key: sum(row.get(key) or 0 for row in rows)
    ratios = {}
    if mode != "dense":
        for name, key in (("pages", "pages"), ("attended", "attended_tokens"), ("elements", "attended_elements"),
                          ("burst", "burst_bytes"), ("hbm", "hbm_bytes"), ("dequant", "dequant_ops")):
            base = sums(dense[b]["layers"], key)
            ratios[name] = sums(snapshot["layers"], key) / base if base else 1.0
    share = ratios.get("pages", 1.0)
    max_reads = max((row.get("max_block_reads") or 0) for row in snapshot["steps"]) if snapshot["steps"] else 0
    hot_step = max_reads / (pages_per_block * share) if share else 1.0
    tpot = {f"b{key}": value["system"][0]["tpot_hbf_est_ns"] for key, value in dense.items() if value["system"]}
    result = {"layer_ns": float(layer_ns), "round_ns": float(round_ns), "balance": kappa, "ratios": ratios,
              "read_share": share, "hot_step": hot_step, "max_block_reads": max_reads, "partition_banks": banks,
              "phase_a_tpot_ns": tpot, "inputs": inputs, "points": len(points)}
    cached.write_text(json.dumps(result, indent=2) + "\n")
    return result

def prefill_tok_s(model, gpu, tokens):
    """Analytic prefill rate: FLOP/token = 2P + 2 L (q_heads d_h) S at the GPU's compute_eff (params.md section 6)."""
    params = model.layers * model.layer_params + model.embedding_params
    flops = 2 * params + 2 * model.layers * model.q_heads * model.head_dim * max(tokens, 1)
    return gpu.flops / flops


def threshold_steps(x, ppb, share=1.0, hot=1.0):
    """Decode steps until a block of fully read KV reaches X reads: X / (ppb x share x hot)."""
    return x / (ppb * share * hot)


def erase_amp(pages, segment_pages):
    """Erase amplification of entries that own whole segments: ceil(P / P_seg) P_seg / P."""
    return math.ceil(pages / segment_pages) * segment_pages / pages if pages > 0 else 1.0


def _pages(geometry, tokens):
    return float(geometry.stored_pages(np.array([int(round(tokens))]))[0])


def analytic(geometry, clock, workload, *, policy="packed", partition="none", pages_per_block=256,
             capacity_pages=512 * GIB // 4096, banks=1024, x_values=(1e6,), pe_limits=(1e5,), share=1.0, hot=1.0,
             retention_s=86400.0, prefix_reads="per_request", t_prog=100000, t_bers=1000000, m5="gaps",
             suspend_ns=None, stacks=1, region_pages=None):
    """Steady-state rows (one per X) for a closed-loop (concurrency) or Poisson (rate) workload.

    Conventions (design-B section 3): pages per request P_r = L H sum_t floor(w_t / tpp_t) (full
    pages); refresh pages per request = floor(s rho / X) x P_r with rho = ppb x share x hot block
    reads per step and s the steps that read the data (a conservative count: generated pages are
    counted as if present from the start); P/E per block per day = programs x erase amp x 86,400 / C.
    """
    page = geometry.page_bytes
    seg_banks = PARTITIONS[partition] if partition != "none" else banks
    seg_pages = seg_banks * pages_per_block
    rho = pages_per_block * share * hot
    w = workload
    n_in, n_out = w.input.mean(), w.output.mean()
    sessions = w.sessions
    calls = w.calls if sessions else 1
    h0 = w.history.mean() if sessions and w.history is not None else 0.0
    prefix = w.prefix_tokens if w.prefix_share else 0
    per_stack = 1.0 / stacks

    def mean_context():
        if sessions:
            return h0 + n_in + (calls - 1) / 2 * (n_in + n_out) + n_out / 2
        return w.prefix_share * prefix + n_in + n_out / 2

    ctx = mean_context()
    hbm_out = policy in ("hbm_decode", "layer_major")
    if w.arrival == "closed":
        active = w.concurrency * per_stack  # sessions or requests per stack
    else:
        active = None
    b = max(1, round(active)) if active is not None else 1
    tpot = None
    for _ in range(30):
        step = clock.step(np.full(b, int(round(ctx))), hbm_tokens=np.full(b, int(n_out / 2)) if hbm_out else None)
        tpot = step.tpot_ns / 1e9
        call_s = n_out * tpot
        if active is not None:
            duty = call_s / (call_s + w.think_s) if sessions else 1.0
            new_b = max(1, round(active * duty))
        else:
            new_b = max(1, round(w.rate * per_stack * calls * call_s))
        if new_b == b:
            break
        b = new_b
    call_s = n_out * tpot
    if active is not None:
        calls_per_s = (active / (call_s + w.think_s)) if sessions else b / call_s
    else:
        calls_per_s = w.rate * per_stack * calls
    entry_rate = calls_per_s / calls  # requests or sessions per second (per stack)
    tokens_per_s = calls_per_s * n_out

    # Entries written per request (or session), split into chunks with their read exposure.
    chunks = []  # (pages, cause, reads per page over its life, lifetime s)
    life_tail = w.retain_s
    if sessions:
        history = 0.0
        for k in range(calls):
            fresh = (h0 + n_in) if k == 0 else n_in
            pre = _pages(geometry, history + fresh) - _pages(geometry, history)
            dec = _pages(geometry, history + fresh + n_out) - _pages(geometry, history + fresh)
            later = calls - 1 - k
            reads_pre = (later + 1) * n_out * rho + later * pages_per_block
            reads_dec = reads_pre - n_out / 2 * rho
            life = (later + 1) * (call_s + w.think_s) + life_tail
            chunks.append((pre, "prefill", reads_pre, life))
            if not hbm_out:
                chunks.append((dec, "decode", reads_dec, life))
            history += fresh + n_out
        entry_pages = sum(c[0] for c in chunks)
    else:
        base = w.prefix_share * _pages(geometry, prefix) if prefix else 0.0
        own_prompt = (w.prefix_share * (_pages(geometry, prefix + n_in) - _pages(geometry, prefix))
                      + (1 - w.prefix_share) * _pages(geometry, n_in))
        full = (w.prefix_share * (_pages(geometry, prefix + n_in + n_out) - _pages(geometry, prefix))
                + (1 - w.prefix_share) * _pages(geometry, n_in + n_out))
        dec = 0.0 if hbm_out else full - own_prompt
        chunks.append((own_prompt, "prefill", n_out * rho, call_s + life_tail))
        if dec:
            chunks.append((dec, "decode", n_out * rho, call_s + life_tail))
        entry_pages = own_prompt + dec
        del base
    host_pages = sum(c[0] for c in chunks)

    # Static batching waves rewrite the whole reserved region (padding included) per wave.
    wave_pages = 0.0
    if policy == "layer_major":
        region = region_pages if region_pages is not None else b * _pages(geometry, n_in) * 1.0
        waves_per_s = calls_per_s / b
        wave_pages = waves_per_s * max(0.0, region - b * host_pages)

    rows = []
    for x in x_values:
        rd, ret, numerator = 0.0, 0.0, 0.0
        for pages, cause, reads, life in chunks:
            rd += math.floor(reads / x) * pages
            numerator += reads * pages
            ret += math.floor(life / retention_s) * pages if retention_s else 0.0
        prefix_rd = prefix_ret = 0.0
        if prefix and w.prefix_groups:
            p_pages = _pages(geometry, prefix)
            weights = 1.0 / np.arange(1, w.prefix_groups + 1) ** w.zipf
            weights /= weights.sum()
            for p in weights:
                readers = b * w.prefix_share * p
                k = readers if prefix_reads == "per_request" else min(1.0, readers)
                per_s = k * rho / tpot + calls_per_s * w.prefix_share * p * pages_per_block
                prefix_rd += p_pages * per_s / x
                numerator += p_pages * per_s / entry_rate
            prefix_ret = (w.prefix_groups * p_pages / retention_s) if retention_s else 0.0
        padding = (seg_banks - 1) / 2 if policy in ("packed", "hbm_decode", "vllm_pool") else 0.0
        rates = {"prefill": entry_rate * sum(c[0] for c in chunks if c[1] == "prefill"),
                 "decode": entry_rate * sum(c[0] for c in chunks if c[1] == "decode"),
                 "refresh_rd": entry_rate * rd + prefix_rd, "refresh_ret": entry_rate * ret + prefix_ret,
                 "compaction": 0.0, "replay": 0.0, "wave": 0.0, "padding": entry_rate * padding + wave_pages}
        if policy == "layer_major":  # refresh of a static region is a replay in place
            rates["replay"], rates["refresh_rd"], rates["refresh_ret"] = (
                rates["refresh_rd"] + rates["refresh_ret"], 0.0, 0.0)
        programs = sum(rates.values())
        amp = erase_amp(entry_pages, seg_pages) if policy == "dedicated" else 1.0
        if policy == "layer_major":
            region = region_pages if region_pages is not None else b * host_pages
            amp = erase_amp(region, seg_pages)
        pe_day = programs * amp * DAY / capacity_pages
        kv = rates["prefill"] + rates["decode"]
        step = clock.step(np.full(b, int(round(ctx))), hbm_tokens=np.full(b, int(n_out / 2)) if hbm_out else None)
        m5_row = {"extra_step_ns": 0.0, "max_step_extra_ns": 0.0}
        if m5 != "off":
            m5_row = interference({"programs": programs * tpot, "erases": programs * amp / pages_per_block * tpot,
                                   "wave_banks": seg_banks, "gaps_ns": step.gaps_ns, "step_ns": step.tpot_ns},
                                  m5, t_prog, t_bers, suspend_ns)
        row = {"x": x, "pages_per_block": pages_per_block, "policy": policy, "partition": partition,
               "workload": w.name, "mean_batch": b, "mean_context": ctx, "mean_tpot_ms": tpot * 1e3,
               "tpot_m5_ms": (step.tpot_ns + m5_row["extra_step_ns"]) / 1e6,
               "max_step_extra_ms": m5_row["max_step_extra_ns"] / 1e6, "req_per_s": calls_per_s,
               "tokens_per_s": tokens_per_s, "pages_per_request": host_pages,
               "kv_bytes_per_token": geometry.token_bytes,
               "threshold_steps": threshold_steps(x, pages_per_block, share, hot),
               "refreshes_per_request": (rd / host_pages) if host_pages else 0.0,
               "kv_write_gb_s": kv * page / 1e9}
        row |= {f"{cause}_gb_s": rates[cause] * page / 1e9 for cause in CAUSES}
        row |= {"program_gb_s": programs * page / 1e9,
                "program_bw_share": programs * page / (banks * page / (t_prog / 1e9)),
                "wa": programs / kv if kv else None, "erase_wa": amp * programs / kv if kv else None,
                "refresh_to_kv": (rates["refresh_rd"] + rates["refresh_ret"] + rates["replay"]) / kv if kv else None,
                "erases_per_s": programs * amp / pages_per_block, "avg_pec_per_day": pe_day,
                "max_pec_per_day": pe_day, "break_even_x": numerator * entry_rate / kv if kv else None}
        for limit in pe_limits:
            row[f"days_to_limit_avg@{limit:g}"] = limit / pe_day if pe_day else math.inf
        row["pe_per_mtok"] = pe_day / (tokens_per_s * DAY / 1e6) if tokens_per_s else None
        partial = geometry.partial_pages() * page
        row["staging_peak_mib"] = (b * partial + seg_banks * page + (
            b * n_out * geometry.token_bytes if hbm_out else 0)) / 2 ** 20
        row["pending_units_per_channel"] = b * geometry.partial_pages() / 16
        row["unit_open_s"] = max(geometry.tpp) * tpot
        row["replay_buffer_mib"] = seg_pages * page / 2 ** 20 if policy == "layer_major" else 0.0
        rows.append(row)
    return rows

# vLLM free-list legality tracker

class VllmTracker:
    """Write legality of a vLLM-style per-layer pool on HBF (one layer's pool; all layers are alike).

    Block id i holds pool pages [i P_id, (i + 1) P_id); a page's NAND block is (lpa mod banks,
    (lpa // banks) // ppb) and its position (lpa // banks) mod ppb. A write is legal when it is
    the block's next position, or position 0 of a block without live pages (auto-erase, section
    5.4.1 item 5). Otherwise it is an overlap (an already-written position, section 11.5.2.2), a
    skip (section 11.5.2.6) or a live erase (page 0 of a block that still holds live data).
    Freed ids return to the free queue (``reuse``: fifo as vLLM, or lifo); stranded ids are free
    ids below the fill position of a block that still holds live data.
    """

    def __init__(self, ids, pages_per_id, pages_per_block, banks=1024, order="fresh", reuse="fifo"):
        from collections import deque
        self.ids, self.pages_per_id, self.ppb, self.banks = ids, pages_per_id, pages_per_block, banks
        rows = -(-ids * pages_per_id // banks)
        self.nblocks = -(-rows // pages_per_block)
        self.fill = np.zeros(banks * self.nblocks, dtype=np.int64)
        self.live = np.zeros(banks * self.nblocks, dtype=np.int64)
        start = np.arange(ids)
        if order.startswith("shuffled:"):
            np.random.Generator(np.random.PCG64(int(order[9:]))).shuffle(start)
        elif order != "fresh":
            raise ValueError("order must be fresh or shuffled:<seed>")
        if reuse not in ("fifo", "lifo"):
            raise ValueError("reuse must be fifo or lifo")
        self.queue, self.reuse = deque(start.tolist()), reuse
        self.used = np.zeros(ids, dtype=bool)
        self.counts = {"skip": 0, "overlap": 0, "live_erase": 0, "erases": 0, "writes": 0, "legalize_pages": 0}

    def _pages(self, ids):
        lpa = (np.asarray(ids, dtype=np.int64)[:, None] * self.pages_per_id + np.arange(self.pages_per_id)).ravel()
        row = lpa // self.banks
        return (lpa % self.banks) * self.nblocks + row // self.ppb, row % self.ppb

    def allocate(self, n):
        if n > len(self.queue):
            raise RuntimeError("vLLM pool exhausted")
        ids = [self.queue.popleft() for _ in range(n)]
        self.used[ids] = True
        self.write(ids)
        return ids

    def write(self, ids):
        if not len(ids):
            return
        key, pos = self._pages(ids)
        owner = np.repeat(np.arange(len(ids)), self.pages_per_id)
        order = np.argsort(key, kind="stable")
        k, p, o = key[order], pos[order], owner[order]
        first = np.ones(len(k), dtype=bool)
        first[1:] = k[1:] != k[:-1]
        prev = np.empty(len(k), dtype=np.int64)
        prev[1:] = p[:-1]
        prev[first] = self.fill[k[first]] - 1
        live_before = self.live[k]
        erase = (p == 0) & first & (live_before == 0)
        legal = erase | (p == prev + 1)
        live_erase = ~legal & (p == 0)
        overlap = ~legal & ~live_erase & (p <= prev)
        skip = ~legal & (p > prev + 1)
        for name, mask in (("overlap", overlap), ("skip", skip), ("live_erase", live_erase)):
            self.counts[name] += len(np.unique(o[mask]))
        bad = np.unique(o[overlap | live_erase])
        if len(bad):  # legalising = replaying the touched blocks up to their fill
            touched = np.unique(k[np.isin(o, bad)])
            self.counts["legalize_pages"] += int(self.fill[touched].sum())
        self.counts["erases"] += int(erase.sum())
        self.counts["writes"] += len(ids)
        ends = np.flatnonzero(np.r_[first[1:], True])
        starts = np.flatnonzero(first)
        group_max = np.maximum.reduceat(p, starts) + 1
        blocks = k[starts]
        reset = np.logical_or.reduceat(erase, starts)
        self.fill[blocks] = np.where(reset, group_max, np.maximum(self.fill[blocks], group_max))
        np.add.at(self.live, k, 1)
        del ends

    def free(self, ids):
        if not len(ids):
            return
        key, _ = self._pages(ids)
        np.subtract.at(self.live, key, 1)
        self.used[list(ids)] = False
        if self.reuse == "fifo":
            self.queue.extend(ids)
        else:
            self.queue.extendleft(ids)

    def stranded(self):
        """Fraction of all ids that are free but cannot be rewritten legally now."""
        free = np.flatnonzero(~self.used)
        if not len(free):
            return 0.0
        key, pos = self._pages(free)
        stuck = (self.live[key] > 0) & (pos < self.fill[key])
        per_id = stuck.reshape(len(free), self.pages_per_id).any(axis=1)
        return float(per_id.sum()) / self.ids

# Event simulation (M4b)

@dataclass
class Device:
    """Capacity and geometry of one stack for the simulator."""
    banks: int = 1024
    pages_per_block: int = 256
    page_bytes: int = 4096
    capacity_pages: int = 512 * GIB // 4096
    partition: str = "none"
    segments: int | None = None  # overrides capacity_pages (tests)

    @property
    def parts(self):
        return 1 if self.partition == "none" else self.banks // PARTITIONS[self.partition]

    @property
    def segment_banks(self):
        return self.banks // self.parts

    @property
    def segment_pages(self):
        return self.segment_banks * self.pages_per_block

    @property
    def count(self):
        return self.segments if self.segments is not None else self.capacity_pages // self.segment_pages


@dataclass
class Options:
    """Simulation policy settings (see the CLI help)."""
    policy: str = "packed"
    wear: str = "dynamic"
    refresh: str = "relocate"
    refresh_idle: str = "relocate"  # or evict
    reclaim: str = "evict"  # or compact
    prefix_reads: str = "per_request"
    x: float = 1e6
    hot: float = 1.0
    share: float = 1.0
    retention_s: float | None = 86400.0
    hours: float = 24.0
    warmup_h: float = 1.0
    quantum_s: float = 1.0
    report_s: float = 60.0
    align: str = "row"
    low_watermark: int | None = None
    m5: str = "gaps"
    t_prog: float = 100000.0
    t_bers: float = 1000000.0
    suspend_ns: float | None = None
    block_unit: str = "die"
    prefill: str = "disaggregated"
    pe_limits: tuple = (1e4, 1e5, 1e6)
    block_tokens: int | None = None
    free_order: str = "fresh"
    vllm_reuse: str = "fifo"
    slots: int | None = None  # layer_major: slots per wave (default: concurrency or max_batch)

    def __post_init__(self):
        if self.policy not in POLICIES:
            raise ValueError(f"policy must be one of {', '.join(POLICIES)}")
        if self.wear not in WEAR or self.refresh not in REFRESH:
            raise ValueError(f"wear must be one of {WEAR} and refresh one of {REFRESH}")
        if self.refresh_idle not in ("relocate", "evict") or self.reclaim not in ("evict", "compact"):
            raise ValueError("refresh_idle is relocate or evict; reclaim is evict or compact")
        if self.prefix_reads not in ("per_request", "once") or self.align not in ("row", "none"):
            raise ValueError("prefix_reads is per_request or once; align is row or none")
        if self.m5 not in M5_MODES + ("off",) or self.prefill not in ("disaggregated", "colocated"):
            raise ValueError(f"m5 must be one of {M5_MODES + ('off',)}; prefill disaggregated or colocated")
        if self.x <= 0 or self.hot <= 0 or not 0 < self.share <= 1 or self.hours <= 0:
            raise ValueError("x, hot and hours must be positive and share in (0, 1]")


class Running:
    __slots__ = ("req", "entry", "prefix", "start", "gen", "own0")

    def __init__(self, req, entry, prefix, start, own0):
        self.req, self.entry, self.prefix, self.start, self.gen, self.own0 = req, entry, prefix, start, 0, own0

    @property
    def remaining(self):
        return self.req.output - self.gen


class Simulator:
    """Request-level simulation at segment granularity (design-B section 4); see ``simulate``."""

    def __init__(self, geometry, clock, workload, device=None, options=None, model=None, gpu=None):
        self.geo, self.clock, self.workload = geometry, clock, workload
        self.dev, self.opt = device or Device(), options or Options()
        self.model, self.gpu = model, gpu
        o, d = self.opt, self.dev
        if o.policy == "layer_major" and o.refresh != "replay":
            self.opt = o = Options(**(asdict(o) | {"refresh": "replay"}))
        self.S, self.P = d.count, d.segment_pages
        if self.S < 2:
            raise ValueError("The device needs at least two segments")
        S = self.S
        self.fill = np.zeros(S)
        self.live = np.zeros(S)
        self.pec = np.zeros(S, dtype=np.int64)
        self.reads = np.zeros(S)
        self.owner = np.full(S, -1, dtype=np.int64)  # dedicated entry, -2 for a wave region
        self.is_open = np.zeros(S, dtype=bool)
        self.part_of = np.arange(S) % d.parts
        self.erases = np.zeros((S, len(CAUSES)), dtype=np.int64)
        self.frontier = [-1] * d.parts
        self.stream = Stream(workload)
        self.t = 0.0
        # entries
        cap = 1024
        self.e_pages = np.zeros(cap)
        self.e_readers = np.zeros(cap)
        self.e_state = np.zeros(cap, dtype=np.int8)  # 0 free, 1 active, 2 idle
        self.e_ttl = np.full(cap, math.inf)
        self.e_part = np.zeros(cap, dtype=np.int64)
        self.e_seg = np.full(cap, -1, dtype=np.int64)
        self.e_tokens = np.zeros(cap)
        self.e_kind = np.zeros(cap, dtype=np.int8)  # 0 private, 1 prefix, 2 session
        self.e_new = np.zeros(cap, dtype=bool)
        self.free_entries = list(range(cap - 1, -1, -1))
        self.e_rows, self.e_segs, self.e_key, self.e_ids = {}, {}, {}, {}
        self.prefix_entry, self.session_entry = {}, {}
        # occupancy rows
        rows = 4096
        self.c_e = np.zeros(rows, dtype=np.int64)
        self.c_s = np.zeros(rows, dtype=np.int64)
        self.c_n = np.zeros(rows)
        self.c_t = np.zeros(rows)
        self.c_v = np.zeros(rows, dtype=bool)
        self.c_count = 0
        self.row_of = {}
        # staging (decode flush rows) per partition (packed) or per entry (dedicated)
        self.staged = [dict() for _ in range(d.parts)]
        self.staged_total = [0.0] * d.parts
        self.running, self.queue = [], []
        self.counters = {"programs": dict.fromkeys(CAUSES, 0.0), "seg_erases": dict.fromkeys(CAUSES, 0),
                         "block_erases": 0, "tokens": 0, "steps": 0, "decode_s": 0.0, "tpot_s": 0.0,
                         "m5_s": 0.0, "batch_s": 0.0, "evicted_pages": 0.0, "recompute_tokens": 0.0,
                         "completed": 0, "refresh_events": 0, "replays": 0, "time": 0.0, "stalls": 0,
                         "stranded_s": 0.0, "prefill_s": 0.0}
        self.max_step_extra = 0.0
        self.staging_peak = 0.0
        self.m5_extra = 0.0
        self.last_step = None
        self.timeseries = []
        self.warm = None
        self.guard = False
        self.vllm = None
        self.region = []
        self.wave = False
        if o.policy == "vllm_pool":
            bt = o.block_tokens or max(self.geo.tpp)
            if any(bt % tpp for tpp in self.geo.tpp):
                raise ValueError("vllm_pool block_tokens must be a multiple of every HBF tensor's tokens per page")
            self.block_tokens = bt
            per_id = self.geo.heads * sum(bt // tpp for tpp in self.geo.tpp)
            pool_pages = d.count * d.segment_pages // self.geo.layers
            self.vllm = VllmTracker(pool_pages // per_id, per_id, d.pages_per_block, d.banks, o.free_order,
                                    o.vllm_reuse)
        max_pages = self._max_request_pages()
        self.low_wm = o.low_watermark if o.low_watermark is not None else 2 + math.ceil(max_pages / self.P)

    # ---- helpers -------------------------------------------------------------------------
    def _max_request_pages(self):
        w = self.workload
        if w.script:
            return max(float(self.geo.stored_pages(np.array([int(r["input"]) + int(r["output"])]))[0])
                       for r in w.script)
        ins = w.input.get("max") or w.input.get("n") or w.input.mean() * 4
        out = w.output.get("max") or w.output.get("n") or w.output.mean() * 4
        hist = (w.history.get("n") or w.history.mean() * 2) if w.history is not None else 0
        calls = max(1, w.calls)
        return float(self.geo.stored_pages(np.array([int(hist + calls * (ins + out) + w.prefix_tokens)]))[0])

    def P_of(self, tokens):
        return float(self.geo.stored_pages(np.array([int(tokens)]))[0])

    def _entry(self, kind, part, tokens):
        if not self.free_entries:
            old = len(self.e_pages)
            new = old * 2
            for name in ("e_pages", "e_readers", "e_ttl", "e_part", "e_seg", "e_tokens", "e_kind", "e_state", "e_new"):
                array = getattr(self, name)
                grown = np.zeros(new, dtype=array.dtype)
                grown[:old] = array
                if name == "e_ttl":
                    grown[old:] = math.inf
                if name == "e_seg":
                    grown[old:] = -1
                setattr(self, name, grown)
            self.free_entries = list(range(new - 1, old - 1, -1))
        e = self.free_entries.pop()
        self.e_pages[e], self.e_readers[e], self.e_state[e], self.e_ttl[e] = 0, 0, 1, math.inf
        self.e_part[e], self.e_seg[e], self.e_tokens[e], self.e_kind[e], self.e_new[e] = part, -1, tokens, kind, True
        self.e_rows[e], self.e_segs[e] = set(), set()
        return e

    def _row(self, e, s, n, t):
        key = (e, s)
        index = self.row_of.get(key)
        if index is not None:
            self.c_n[index] += n
            return
        if self.c_count == len(self.c_e):
            valid = int(self.c_v[:self.c_count].sum())
            if valid < self.c_count // 2:
                self._compact_rows()
            else:
                for name in ("c_e", "c_s", "c_n", "c_t", "c_v"):
                    array = getattr(self, name)
                    grown = np.zeros(len(array) * 2, dtype=array.dtype)
                    grown[:len(array)] = array
                    setattr(self, name, grown)
        index = self.c_count
        self.c_e[index], self.c_s[index], self.c_n[index], self.c_t[index], self.c_v[index] = e, s, n, t, True
        self.c_count += 1
        self.row_of[key] = index
        self.e_rows[e].add(index)

    def _compact_rows(self):
        keep = np.flatnonzero(self.c_v[:self.c_count])
        for name in ("c_e", "c_s", "c_n", "c_t", "c_v"):
            array = getattr(self, name)
            array[:len(keep)] = array[keep]
        self.c_v[len(keep):] = False
        self.c_count = len(keep)
        self.row_of = {(int(self.c_e[i]), int(self.c_s[i])): i for i in range(self.c_count)}
        self.e_rows = {e: set() for e in self.e_rows}
        for i in range(self.c_count):
            self.e_rows.setdefault(int(self.c_e[i]), set()).add(i)

    def _drop_row(self, index):
        e, s = int(self.c_e[index]), int(self.c_s[index])
        n = self.c_n[index]
        self.c_v[index] = False
        self.row_of.pop((e, s), None)
        self.e_rows[e].discard(index)
        self.live[s] -= n
        self.e_pages[e] -= n
        if self.live[s] < 1e-9:
            self.live[s] = 0.0
        return e, s, n

    def _free_mask(self, part=None):
        mask = (self.live <= 0) & ~self.is_open & (self.owner == -1)
        return mask if part is None else mask & (self.part_of == part)

    def _pick(self, part):
        free = np.flatnonzero(self._free_mask(part))
        if not len(free):
            return -1
        if self.opt.wear == "none":
            return int(free[0])
        return int(free[np.argmin(self.pec[free])])

    def _erase(self, s, cause):
        """Row 0 write of segment s: auto-erase of its blocks (every page-0 write counts)."""
        self.pec[s] += 1
        self.erases[s, CAUSES.index(cause)] += 1
        self.counters["seg_erases"][cause] += 1
        self.counters["block_erases"] += self.dev.segment_banks
        self.reads[s] = 0.0
        self.fill[s] = 0.0

    def _open(self, part, cause, owner=-1):
        s = self._pick(part)
        if s < 0:
            if not self.guard:
                self._reclaim(part, force=True)
            s = self._pick(part)
            if s < 0:
                raise RuntimeError("HBF capacity exhausted: no free segment even after reclaim")
        self._erase(s, cause)
        self.is_open[s] = True
        self.owner[s] = owner
        if not self.guard:
            self._reclaim(part)
        return s

    def _program(self, cause, n):
        self.counters["programs"][cause] += n
        self.programs_q += n

    # ---- writes --------------------------------------------------------------------------
    def _write(self, e, n, cause, align=False):
        """Append n pages of entry e (packed frontier of its partition, or its own segments)."""
        if n <= 0:
            return
        part = int(self.e_part[e])
        dedicated = self.opt.policy == "dedicated" or (self.e_kind[e] == 1 and self.opt.policy == "dedicated")
        while n > 1e-9:
            if dedicated:
                s = int(self.e_seg[e])
                if s < 0 or self.fill[s] >= self.P - 1e-9:
                    if s >= 0:
                        self.is_open[s] = False
                    s = self._open(part, cause if cause != "padding" else "prefill", owner=e)
                    self.e_seg[e] = s
                    self.e_segs[e].add(s)
            else:
                s = self.frontier[part]
                if s >= 0 and align and self.opt.align == "row":
                    pad = (-self.fill[s]) % self.dev.segment_banks
                    if pad > 1e-9 and self.fill[s] + pad <= self.P + 1e-9:
                        self.fill[s] += pad
                        self._program("padding", pad)
                    align = False
                if s < 0 or self.fill[s] >= self.P - 1e-9:
                    if s >= 0:
                        self.is_open[s] = False
                    self.frontier[part] = -1
                    s = self._open(part, cause)
                    self.frontier[part] = s
                    align = False
            k = min(n, self.P - self.fill[s])
            self.fill[s] += k
            self.live[s] += k
            self.e_pages[e] += k
            self._row(e, s, k, self.t)
            self._program(cause, k)
            n -= k

    def _stage(self, e, pages):
        if pages <= 0:
            return
        if self.opt.policy == "dedicated":
            key, part = e, int(self.e_part[e])
            book = self.staged[part]
            book[key] = book.get(key, 0.0) + pages
            row = self.dev.segment_banks
            if book[key] >= row:
                whole = math.floor(book[key] / row) * row
                book[key] -= whole
                self._write(e, whole, "decode")
            return
        part = int(self.e_part[e])
        book = self.staged[part]
        book[e] = book.get(e, 0.0) + pages
        self.staged_total[part] += pages
        row = self.dev.segment_banks
        if self.staged_total[part] >= row:
            whole = math.floor(self.staged_total[part] / row) * row
            self.staged_total[part] -= whole
            for key in list(book):
                if whole <= 1e-9:
                    break
                take = min(book[key], whole)
                book[key] -= take
                whole -= take
                if book[key] <= 1e-9:
                    del book[key]
                self._write(key, take, "decode")

    def _unstage(self, e, write):
        part = int(self.e_part[e])
        pages = self.staged[part].pop(e, 0.0)
        if self.opt.policy != "dedicated":
            self.staged_total[part] -= pages
        if write and pages > 0:
            self._write(e, pages, "decode")

    # ---- entry life ----------------------------------------------------------------------
    def _kill(self, e, evicted=False):
        if evicted:
            self.counters["evicted_pages"] += self.e_pages[e]
            self.counters["recompute_tokens"] += self.e_tokens[e]
        self._unstage(e, write=False)
        for index in list(self.e_rows.get(e, ())):
            self._drop_row(index)
        for s in self.e_segs.get(e, ()):
            if self.owner[s] == e:
                self.owner[s] = -1
                self.is_open[s] = False
        if self.vllm is not None and e in self.e_ids:
            self.vllm.free(self.e_ids.pop(e))
        key = self.e_key.pop(e, None)
        if key is not None:
            table = self.prefix_entry if key[0] == "prefix" else self.session_entry
            if table.get(key[1]) == e:
                del table[key[1]]
        self.e_rows.pop(e, None)
        self.e_segs.pop(e, None)
        self.e_state[e], self.e_pages[e], self.e_readers[e] = 0, 0.0, 0
        self.free_entries.append(e)

    def _idle(self, e, ttl):
        self.e_state[e] = 2
        self.e_readers[e] = 0
        self.e_ttl[e] = ttl
        self._unstage(e, write=True)

    def _read_once(self, e):
        """Prefill attention over a cached entry: every page read once."""
        rows = [i for i in self.e_rows.get(e, ()) if self.c_v[i]]
        if rows:
            np.add.at(self.reads, self.c_s[rows], self.c_n[rows] / self.dev.segment_banks)

    def _part_for(self):
        if self.dev.parts == 1:
            return 0
        load = np.bincount(self.part_of, weights=self.live, minlength=self.dev.parts)
        return int(np.argmin(load))

    def _space_ok(self, pages, part):
        free = int(self._free_mask(part).sum())
        room = free * self.P
        s = self.frontier[part]
        if s >= 0:
            room += self.P - self.fill[s]
        if room - pages >= (self.low_wm - 1) * self.P:
            return True
        self._reclaim(part, force=True)
        free = int(self._free_mask(part).sum())
        return free * self.P + (self.P - self.fill[s] if s >= 0 else 0) >= pages

    def _admit(self, req, now):
        """Start a request: prefix and session entries, prefill writes. False when space is short."""
        o, w = self.opt, self.workload
        prefix_tokens = w.prefix_tokens if req.prefix >= 0 else 0
        start = prefix_tokens + req.history + req.input
        need = self.P_of(start + req.output) - self.P_of(prefix_tokens + req.history)
        if o.policy in ("hbm_decode",):
            need = self.P_of(start) - self.P_of(prefix_tokens + req.history)
        part = self._part_for()
        if o.policy != "layer_major" and not self._space_ok(need, part):
            self.counters["stalls"] += 1
            return False
        self.t = now
        pre = -1
        if req.prefix >= 0:
            pre = self.prefix_entry.get(req.prefix, -1)
            if pre >= 0:
                self._read_once(pre)
            else:
                pre = self._entry(1, part, prefix_tokens)
                self.prefix_entry[req.prefix] = pre
                self.e_key[pre] = ("prefix", req.prefix)
                self._write(pre, self.P_of(prefix_tokens), "prefill", align=True)
                self._vllm_write(pre, 0, prefix_tokens)
            self.e_state[pre] = 1
            self.e_readers[pre] += 1
            self.e_ttl[pre] = math.inf
        if req.session >= 0 and req.session in self.session_entry:
            e = self.session_entry[req.session]
            self._read_once(e)
            self.e_state[e], self.e_readers[e], self.e_ttl[e] = 1, 1, math.inf
            own0 = req.history
        else:
            e = self._entry(2 if req.session >= 0 else 0, part, 0)
            if req.session >= 0:
                self.session_entry[req.session] = e
                self.e_key[e] = ("session", req.session)
            self.e_readers[e] = 1
            own0 = prefix_tokens
            if req.history:  # an evicted session history is recomputed and rewritten
                self.counters["recompute_tokens"] += req.history
                own0 = prefix_tokens
        written_from = own0 if not (req.session >= 0 and own0 == req.history) else req.history
        pages = self.P_of(start) - self.P_of(written_from)
        self.e_tokens[e] = start - prefix_tokens
        self._write(e, pages, "prefill", align=bool(self.e_new[e]))
        self._vllm_write(e, written_from - prefix_tokens, start - prefix_tokens)
        self.e_new[e] = False
        if o.prefill == "colocated" and self.model is not None:
            self.prefill_time += req.input / prefill_tok_s(self.model, self.gpu, start)
        self.running.append(Running(req, e, pre, start, written_from))
        return True

    def _vllm_write(self, e, lo, hi):
        """vllm_pool: allocate and write the ids of the full vLLM blocks of tokens [lo, hi) of entry e."""
        if self.vllm is None:
            return
        bt = self.block_tokens
        count = hi // bt - lo // bt
        if count > 0:
            self.e_ids.setdefault(e, []).extend(self.vllm.allocate(count))

    def _complete(self, run, now):
        req, e = run.req, run.entry
        self.counters["completed"] += 1
        if run.prefix >= 0:
            self.e_readers[run.prefix] -= 1
            if self.e_readers[run.prefix] <= 0:
                if self.workload.retain_s > 0:
                    self._idle(run.prefix, now + self.workload.retain_s)
                else:
                    self._kill(run.prefix)
        if req.session >= 0 and not req.last:
            self._idle(e, math.inf)
        elif req.retain_s > 0:
            self._idle(e, now + req.retain_s)
        else:
            self._kill(e)
        self.stream.call_done(req, now)

    # ---- read disturb, retention, reclaim -------------------------------------------------
    def _entry_rates(self):
        rate = np.zeros(len(self.e_pages))
        active = self.e_state == 1
        readers = self.e_readers.copy()
        if self.opt.prefix_reads == "once":
            readers = np.minimum(readers, 1)
        rate[active] = readers[active] * self.opt.share
        return rate

    def _seg_rates(self, rate):
        n = self.c_count
        v = self.c_v[:n]
        if not v.any():
            return np.zeros(self.S)
        e, s, pages = self.c_e[:n][v], self.c_s[:n][v], self.c_n[:n][v]
        return np.bincount(s, pages * rate[e], self.S) / self.dev.segment_banks * self.opt.hot

    def _accrue(self, steps_by_entry):
        n = self.c_count
        v = self.c_v[:n]
        if not v.any():
            return
        e, s, pages = self.c_e[:n][v], self.c_s[:n][v], self.c_n[:n][v]
        add = np.bincount(s, pages * steps_by_entry[e] * self.opt.share, self.S)
        self.reads += add / self.dev.segment_banks * self.opt.hot

    def _refresh(self, s, cause):
        self.counters["refresh_events"] += 1
        if self.opt.refresh == "replay":
            self.counters["replays"] += 1
            self._erase(s, "replay")
            self.fill[s] = self.fill_before_replay(s)
            self._program("replay", self.fill[s])
            rows = np.flatnonzero(self.c_v[:self.c_count] & (self.c_s[:self.c_count] == s))
            self.c_t[rows] = self.t
            return
        rows = np.flatnonzero(self.c_v[:self.c_count] & (self.c_s[:self.c_count] == s))
        if self.is_open[s]:
            self.is_open[s] = False
            if self.frontier[self.part_of[s]] == s:
                self.frontier[self.part_of[s]] = -1
        moves = []
        for index in rows:
            e = int(self.c_e[index])
            if self.e_state[e] == 2 and self.opt.refresh_idle == "evict":
                self._kill(e, evicted=True)
                continue
            if not self.c_v[index]:
                continue
            _, _, n = self._drop_row(index)
            moves.append((e, n))
        self.reads[s] = 0.0 if self.live[s] <= 0 else self.reads[s]
        for e, n in moves:
            if self.e_seg[e] == s:
                self.e_seg[e] = -1
            self._write(e, n, cause)

    def fill_before_replay(self, s):
        return self.replay_fill.get(s, self.P)

    def _retention_due(self):
        if not self.opt.retention_s:
            return []
        n = self.c_count
        v = self.c_v[:n]
        if not v.any():
            return []
        oldest = np.full(self.S, math.inf)
        np.minimum.at(oldest, self.c_s[:n][v], self.c_t[:n][v])
        return np.flatnonzero(oldest <= self.t - self.opt.retention_s + 1e-9).tolist()

    def _next_retention(self):
        if not self.opt.retention_s:
            return math.inf
        n = self.c_count
        v = self.c_v[:n]
        return float(self.c_t[:n][v].min()) + self.opt.retention_s if v.any() else math.inf

    def _reclaim(self, part, force=False):
        """Reclaim segments of a partition while free segments < the low watermark (greedy victim)."""
        if self.opt.policy in ("dedicated", "layer_major") or self.guard:
            return
        self.guard = True
        try:
            for _ in range(self.S):
                free = int(self._free_mask(part).sum())
                if free >= self.low_wm and not force:
                    break
                force = False
                candidates = np.flatnonzero((self.part_of == part) & ~self.is_open & (self.live > 0)
                                            & (self.live < self.P - 1e-9) & (self.owner == -1))
                if not len(candidates):
                    break
                n = self.c_count
                v = self.c_v[:n]
                active = (self.e_state == 1)[self.c_e[:n]] & v
                active_live = np.bincount(self.c_s[:n][active], self.c_n[:n][active], self.S)
                victim = int(candidates[np.argmin(active_live[candidates])])
                if active_live[victim] >= self.P - 1e-9:
                    break
                rows = np.flatnonzero(v & (self.c_s[:n] == victim))
                moves = []
                for index in rows:
                    e = int(self.c_e[index])
                    if not self.c_v[index]:
                        continue
                    if self.e_state[e] == 2 and self.opt.reclaim == "evict":
                        self._kill(e, evicted=True)
                        continue
                    _, _, pages = self._drop_row(index)
                    moves.append((e, pages))
                for e, pages in moves:
                    self._write(e, pages, "compaction")
        finally:
            self.guard = False

    # ---- layer-major waves ---------------------------------------------------------------
    def _region_pages(self, slots, prompt):
        if self.model is not None:
            try:
                from .access import LAYOUTS, Shape
                from .config import load
                cfg = load("ocp", pages_per_block=self.dev.pages_per_block,
                           blocks_per_plane=max(1, 131072 // self.dev.pages_per_block))
                lay = LAYOUTS["striped_contiguous"](cfg, Shape(self.geo.layers, self.geo.heads * slots,
                                                               self.model.head_dim, prompt), tensors=self.geo.tensors)
                return float(lay.span)
            except (ValueError, KeyError):
                pass
        per = sum(-(-(int(hi - lo)) // tpp) for spec, tpp in zip(self.geo.hbf, self.geo.tpp)
                  for lo, hi in [tuple(int(v[0]) for v in self.geo.window(spec, np.array([prompt])))])
        return float(slots * self.geo.layers * self.geo.heads * per)

    def _start_wave(self, now):
        w = self.workload
        slots = self.opt.slots or (w.concurrency if w.arrival == "closed" and not w.script else w.max_batch)
        reqs = list(self.queue[:slots])
        del self.queue[:len(reqs)]
        while len(reqs) < slots:
            new = self.stream.closed_new(now)
            if new is None:
                break
            reqs.append(new)
        if not reqs:
            return
        self.t = now
        prompts = [r.history + r.input for r in reqs]
        span = self._region_pages(slots, max(prompts))
        need = math.ceil(span / self.P)
        free = np.flatnonzero(self._free_mask(0))
        if len(free) < need:
            raise RuntimeError(f"layer_major region of {span:.0f} pages needs {need} segments, {len(free)} free")
        chosen = free[:need] if self.opt.wear == "none" else free[np.argsort(self.pec[free], kind="stable")[:need]]
        self.region = [int(s) for s in chosen]
        left = span
        fills = []
        for s in self.region:
            self._erase(s, "wave")
            self.owner[s] = -2
            self.fill[s] = min(self.P, left)
            fills.append(self.fill[s])
            left -= self.fill[s]
        self.replay_fill = {s: f for s, f in zip(self.region, fills)}
        real = 0.0
        for req, prompt in zip(reqs, prompts):
            e = self._entry(0, 0, prompt)
            pages = self.P_of(prompt)
            real += pages
            for s, f in zip(self.region, fills):
                n = pages * f / span
                self.live[s] += n
                self.e_pages[e] += n
                self._row(e, s, n, now)
            self.e_readers[e] = 1
            self.running.append(Running(req, e, -1, prompt, prompt))
        self._program("prefill", real)
        self._program("padding", span - real)
        self.counters["wave_regions"] = self.counters.get("wave_regions", 0) + 1
        self.wave = True

    def _end_wave(self):
        for s in self.region:
            self.owner[s] = -1
        self.region, self.wave = [], False

    # ---- main loop -----------------------------------------------------------------------
    def _admit_pending(self, now):
        o, w = self.opt, self.workload
        self.queue.extend(self.stream.arrived(now))
        if o.policy == "layer_major":
            if not self.running:
                if self.wave:
                    self._end_wave()
                self._start_wave(now)
            return
        cap = w.concurrency if (w.arrival == "closed" and not w.sessions and not w.script) else w.max_batch
        if w.arrival == "closed" and not w.sessions and not w.script:
            while len(self.running) + len(self.queue) < cap:
                new = self.stream.closed_new(now)
                if new is None:
                    break
                self.queue.append(new)
        while self.queue and len(self.running) < cap:
            if not self._admit(self.queue[0], now):
                break
            self.queue.pop(0)

    def _contexts(self):
        ctx = np.array([r.start + r.gen for r in self.running], dtype=np.int64)
        hbm = None
        if self.opt.policy in ("hbm_decode", "layer_major"):
            hbm = np.array([r.gen for r in self.running], dtype=np.int64)
        parts = None
        if self.dev.parts > 1:
            parts = np.array([self.e_part[r.entry] for r in self.running])
        return ctx, hbm, parts

    def _snapshot(self):
        c = self.counters
        return {"programs": dict(c["programs"]), "seg_erases": dict(c["seg_erases"]),
                **{key: c[key] for key in c if key not in ("programs", "seg_erases")},
                "pec": self.pec.copy(), "t": self.t}

    def _record(self, step):
        o = self.opt
        stranded = self.vllm.stranded() if self.vllm is not None else 0.0
        self.last_stranded = stranded
        c = self.counters
        row = {"t_s": self.t, "running": len(self.running), "queued": len(self.queue),
               "sum_context": int(sum(r.start + r.gen for r in self.running)),
               "tpot_ms": step.tpot_ns / 1e6 if step else None,
               "tpot_m5_ms": (step.tpot_ns + self.m5_extra) / 1e6 if step else None}
        row |= {f"{cause}_gb": c["programs"][cause] * self.dev.page_bytes / 1e9 for cause in CAUSES}
        row |= {f"erases_{cause}": c["seg_erases"][cause] for cause in CAUSES}
        row |= {"free_segments": int(self._free_mask().sum()), "live_pages": float(self.live.sum()),
                "stranded_frac": stranded, "avg_pec": float(self.pec.mean()), "max_pec": int(self.pec.max()),
                "max_reads_over_x": float(self.reads.max() / o.x)}
        self.timeseries.append(row)

    def run(self):
        o, w = self.opt, self.workload
        horizon, warm = o.hours * 3600, o.warmup_h * 3600
        next_report = 0.0
        self.replay_fill = {}
        self.prefill_time = 0.0
        step = None
        while self.t < horizon - 1e-9:
            if self.warm is None and self.t >= warm - 1e-9:
                self.warm = self._snapshot()
            now = self.t
            self.programs_q = 0.0
            erases_before = self.counters["block_erases"]
            self.prefill_time = 0.0
            self._admit_pending(now)
            if self.t >= next_report - 1e-9:
                self._record(step)
                next_report += o.report_s
            if not self.running:
                idle = [self.stream.next_time(), horizon, next_report, self._next_retention()]
                if self.warm is None:
                    idle.append(warm)
                idle_ttl = self.e_ttl[self.e_state == 2]
                if len(idle_ttl):
                    idle.append(float(idle_ttl.min()))
                target = max(min(idle), now + 1e-6)
                self.t = target
                self._housekeeping()
                if self.stream.next_time() == math.inf and not self.queue and self.stream.exhausted() and \
                        not (self.e_state == 2).any():
                    if w.arrival == "closed" or w.script:
                        self.t = max(self.t, min(horizon, next_report)) if next_report < horizon else horizon
                continue
            ctx, hbm, parts = self._contexts()
            step = self.clock.step(ctx, parts, self.dev.segment_banks if parts is not None else None, hbm)
            self.last_step = step
            T = (step.tpot_ns + self.m5_extra) / 1e9
            n = max(1, int(o.quantum_s / T))
            rates = self._seg_rates(self._entry_rates())
            pos = rates > 0
            if pos.any():
                to_x = np.ceil((o.x - self.reads[pos]) / rates[pos] - 1e-9)
                n = max(1, min(n, int(max(1.0, to_x.min()))))
            nxt = self._next_retention()
            if nxt < math.inf:
                n = max(1, min(n, math.ceil(max(nxt - now, 0) / T)))
            arrival = self.stream.next_time()
            if arrival < math.inf and (len(self.running) < w.max_batch or o.policy == "layer_major"):
                n = max(1, min(n, math.ceil(max(arrival - now, 0) / T)))
            if self.queue and o.policy != "layer_major":
                n = max(1, min(n, min(r.remaining for r in self.running)))
            n = max(1, min(n, math.ceil(max(horizon - now, T) / T)))
            self._advance(n, T, now)
            elapsed = n * T + self.prefill_time
            self.t = now + elapsed
            c = self.counters
            c["steps"] += n
            c["decode_s"] += elapsed
            c["tpot_s"] += n * step.tpot_ns / 1e9
            c["m5_s"] += n * self.m5_extra / 1e9
            c["batch_s"] += step.batch * n * T
            c["prefill_s"] += self.prefill_time
            if self.vllm is not None:
                c["stranded_s"] += getattr(self, "last_stranded", 0.0) * elapsed
            for s in np.flatnonzero(self.reads >= o.x - 1e-9).tolist():
                if self.live[s] > 0 or self.opt.refresh == "replay" and self.owner[s] == -2:
                    self._refresh(s, "refresh_rd")
                else:
                    self.reads[s] = 0.0
            self._housekeeping()
            staging = sum(self.staged_total) if o.policy != "dedicated" else sum(
                sum(book.values()) for book in self.staged)
            staging = (staging + len(self.running) * self.geo.partial_pages()) * self.dev.page_bytes
            if o.policy in ("hbm_decode", "layer_major"):
                staging += sum(r.gen for r in self.running) * self.geo.token_bytes
            self.staging_peak = max(self.staging_peak, staging)
            if o.m5 != "off":
                load = {"programs": self.programs_q / n,
                        "erases": (self.counters["block_erases"] - erases_before) / n,
                        "wave_banks": self.dev.segment_banks, "gaps_ns": step.gaps_ns, "step_ns": step.tpot_ns}
                result = interference(load, o.m5, o.t_prog, o.t_bers, o.suspend_ns, block_unit=o.block_unit)
                self.m5_extra = result["extra_step_ns"]
                if self.t >= warm:
                    self.max_step_extra = max(self.max_step_extra, result["max_step_extra_ns"])
        if self.warm is None:
            self.warm = self._snapshot()
        self._record(step)
        return self.summary()

    def _advance(self, n, T, now):
        """n decode steps; closed-loop slots are refilled at the step their request completes."""
        budget = {id(r): n for r in self.running}
        w = self.workload
        refill = w.arrival == "closed" and not w.sessions and not w.script and self.opt.policy != "layer_major"
        while True:
            act = [r for r in self.running if budget.get(id(r), 0) > 0 and r.remaining > 0]
            if not act:
                break
            steps = np.zeros(len(self.e_pages))
            advance = {}
            for r in act:
                adv = min(budget[id(r)], r.remaining)
                advance[id(r)] = adv
                steps[r.entry] += adv
                if r.prefix >= 0:
                    if self.opt.prefix_reads == "once":
                        steps[r.prefix] = max(steps[r.prefix], adv)
                    else:
                        steps[r.prefix] += adv
            self._accrue(steps)
            done = []
            for r in act:
                adv = advance[id(r)]
                before = self.P_of(r.start + r.gen)
                r.gen += adv
                budget[id(r)] -= adv
                self.counters["tokens"] += adv
                if self.opt.policy not in ("hbm_decode", "layer_major"):
                    self._stage(r.entry, self.P_of(r.start + r.gen) - before)
                    self._vllm_write(r.entry, r.start + r.gen - adv - (r.start - self.e_tokens[r.entry]) * 0
                                     - self._own_offset(r), r.start + r.gen - self._own_offset(r))
                if r.remaining <= 0:
                    done.append(r)
            for r in done:
                left = budget.pop(id(r), 0)
                t_done = now + (n - left) * T
                self.t = t_done
                self.running.remove(r)
                self._complete(r, t_done)
                if refill and left > 0:
                    new = self.stream.closed_new(t_done)
                    if new is not None and self._admit(new, t_done):
                        budget[id(self.running[-1])] = left
                    elif new is not None:
                        self.queue.append(new)
            if self.opt.policy == "layer_major" and not self.running and self.wave:
                self._end_wave()
        self.t = now

    def _own_offset(self, r):
        return self.workload.prefix_tokens if r.req.prefix >= 0 else 0

    def _housekeeping(self):
        for s in self._retention_due():
            self._refresh(s, "refresh_ret")
        expired = np.flatnonzero((self.e_state == 2) & (self.e_ttl <= self.t + 1e-9))
        for e in expired.tolist():
            self._kill(int(e))
        for part in range(self.dev.parts):
            self._reclaim(part)

    # ---- results -------------------------------------------------------------------------
    def summary(self):
        o, d, c, w0 = self.opt, self.dev, self.counters, self.warm
        span = max(self.t - w0["t"], 1e-9)
        page = d.page_bytes
        delta = {cause: c["programs"][cause] - w0["programs"][cause] for cause in CAUSES}
        rate = {cause: delta[cause] * page / span / 1e9 for cause in CAUSES}
        host = delta["prefill"] + delta["decode"]
        programs = sum(delta.values())
        erases = c["block_erases"] - w0["block_erases"]
        days = span / DAY
        pec = self.pec - w0["pec"]
        steps = c["steps"] - w0["steps"]
        tokens = c["tokens"] - w0["tokens"]
        avg_day = float(pec.mean()) / days
        max_day = float(pec.max()) / days
        refresh = delta["refresh_rd"] + delta["refresh_ret"] + delta["replay"]
        row = {"policy": o.policy, "partition": d.partition, "workload": self.workload.name, "x": o.x,
               "pages_per_block": d.pages_per_block, "hours": o.hours, "measured_h": span / 3600,
               "tokens_per_s": tokens / span,
               "mean_batch": (c["batch_s"] - w0["batch_s"]) / max(c["decode_s"] - w0["decode_s"], 1e-12),
               "mean_tpot_ms": (c["tpot_s"] - w0["tpot_s"]) / steps * 1e3 if steps else None,
               "tpot_m5_ms": (c["tpot_s"] - w0["tpot_s"] + c["m5_s"] - w0["m5_s"]) / steps * 1e3 if steps else None,
               "max_step_extra_ms": self.max_step_extra / 1e6,
               "kv_write_gb_s": host * page / span / 1e9}
        row |= {f"{cause}_gb_s": rate[cause] for cause in CAUSES}
        row |= {"program_gb_s": programs * page / span / 1e9,
                "program_bw_share": programs * page / span / (d.banks * page / (o.t_prog / 1e9)),
                "wa": programs / host if host else None,
                "erase_wa": erases * d.pages_per_block / host if host else None,
                "refresh_to_kv": refresh / host if host else None,
                "erases_per_s": erases / span, "segment_erases": sum(c["seg_erases"][k] - w0["seg_erases"][k]
                                                                     for k in CAUSES),
                "avg_pec_per_day": avg_day, "max_pec_per_day": max_day}
        for limit in o.pe_limits:
            row[f"days_to_limit_avg@{limit:g}"] = limit / avg_day if avg_day else math.inf
            row[f"days_to_limit_max@{limit:g}"] = limit / max_day if max_day else math.inf
        row["pe_per_mtok"] = avg_day / (tokens / span * DAY / 1e6) if tokens else None
        row["completed"] = c["completed"] - w0["completed"]
        row["refresh_events"] = c["refresh_events"] - w0["refresh_events"]
        row["stranded_frac"] = (c["stranded_s"] - w0["stranded_s"]) / span if self.vllm is not None else 0.0
        row["evicted_gb"] = (c["evicted_pages"] - w0["evicted_pages"]) * page / 1e9
        row["recompute_tokens"] = c["recompute_tokens"] - w0["recompute_tokens"]
        counts = self.vllm.counts if self.vllm is not None else {}
        for name in ("skip", "overlap", "live_erase"):
            row[f"violations_{name}"] = counts.get(name, 0)
        row["legalize_gb"] = counts.get("legalize_pages", 0) * page * self.geo.layers / 1e9
        row["staging_peak_mib"] = self.staging_peak / 2 ** 20
        row["pending_units_per_channel"] = row["mean_batch"] * self.geo.partial_pages() / 16
        row["replay_buffer_mib"] = self.P * page / 2 ** 20 if o.refresh == "replay" else 0.0
        row["stalls"] = c["stalls"] - w0["stalls"]
        row["legal"] = o.policy != "vllm_pool"
        return row

    def segment_rows(self):
        rows = []
        for s in range(self.S):
            row = {"segment": s, "partition": int(self.part_of[s]), "pec": int(self.pec[s]),
                   "live_pages": float(self.live[s]), "fill_pages": float(self.fill[s]), "reads": float(self.reads[s])}
            row |= {f"erases_{cause}": int(self.erases[s, i]) for i, cause in enumerate(CAUSES)}
            rows.append(row)
        return rows


def simulate(geometry, clock, workload, device=None, options=None, model=None, gpu=None):
    """Run the event simulation; returns (summary row, Simulator)."""
    sim = Simulator(geometry, clock, workload, device, options, model, gpu)
    return sim.run(), sim
